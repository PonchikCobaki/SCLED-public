role: begin/ middle/ end
type-gate: strip/ rect/ hex/ triangle/ circle
offset-voltage: -256 to 255 (+- 250mV at range 0-1V)
state: off/ run/ pause/ low_batt
program-type: solid/ gradient/ blink/ wave
gradient-number: 0-43
blend-type: noblend/ linearblend/ linearblend-nowarp
speed: 0-255
hsv: ffffff  color format(hue, saturation, brightness)(255,255,255)
